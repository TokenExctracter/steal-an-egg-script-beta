-- ========== HELPERS ==========
local function tween(obj, props, time, style, dir)
    local info = TweenInfo.new(time or 0.25, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

local function waitForChildSafe(parent, name, timeout)
    timeout = timeout or 5
    local child = parent:FindFirstChild(name)
    if child then return child end
    local start = tick()
    while not child and tick() - start < timeout do
        task.wait(0.1)
        child = parent:FindFirstChild(name)
    end
    return child
end

-- Find all egg objects in workspace (flexible: matches common naming patterns)
local function getEggs()
    local eggs = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local n = obj.Name:lower()
            if n:match("egg") and not n:match("esp") and not n:match("highlight") then
                -- skip if it's an ESP we created
                if not obj:FindFirstChild("__CHINAESP") then
                    table.insert(eggs, obj)
                end
            end
        end
    end
    return eggs
end

-- Find deposit/sell zones (matches "deposit", "sell", "base", "spawn")
local function getDepositZones()
    local zones = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local n = obj.Name:lower()
            if n:match("deposit") or n:match("sell") or n:match("zone") or n:match("base") then
                table.insert(zones, obj)
            end
        end
    end
    return zones
end

-- Find closest target to LocalPlayer
local function closestFrom(list)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local root = char.HumanoidRootPart
    local best, bestDist
    for _, obj in ipairs(list) do
        local part = obj:IsA("Model") and obj.PrimaryPart or (obj:IsA("BasePart") and obj or nil)
        if not part and obj:IsA("Model") then
            part = obj:FindFirstChildWhichIsA("BasePart")
        end
        if part then
            local d = (root.Position - part.Position).Magnitude
            if not bestDist or d < bestDist then
                best, bestDist = obj, d
            end
        end
    end
    return best, bestDist
end

-- Get other living players (excluding self)
local function getOtherPlayers()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            table.insert(list, p)
        end
    end
    return list
end

-- ========== MOVEMENT (BYPASS) ==========
-- Use CFrame teleport in small increments + velocity spoof to bypass WalkSpeed checks
local function walkTo(pos, speed)
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum  = char:FindFirstChild("Humanoid")
    if not root or not hum then return end

    local startPos = root.Position
    local dist = (Vector3.new(pos.X, startPos.Y, pos.Z) - startPos).Magnitude
    if dist < 1 then return end

    local dir = (Vector3.new(pos.X, startPos.Y, pos.Z) - startPos).Unit
    local step = (speed or Config.Speed) * (1/60)

    -- Move in small chunks to dodge anticheat speed flags
    local traveled = 0
    while traveled < dist and State.Running do
        local nextPos = root.Position + dir * math.min(step, dist - traveled)
        root.CFrame = CFrame.new(nextPos, nextPos + dir)
        -- spoof WalkSpeed so server-side checks see normal value
        if hum.WalkSpeed ~= 16 then hum.WalkSpeed = 16 end
        traveled = traveled + step
        task.wait(1/60)
    end
end

local function teleportTo(pos)
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    root.CFrame = CFrame.new(pos)
end

-- ========== ESP SYSTEM ==========
local function setEggESP(on)
    -- clear existing
    for _, h in pairs(State.Highlights) do
        if h and h.Parent then h:Destroy() end
    end
    State.Highlights = {}

    if not on then return end

    task.spawn(function()
        while Config.EggESP do
            for _, egg in ipairs(getEggs()) do
                local part = egg:IsA("Model") and egg.PrimaryPart or egg
                if part then
                    if not part:FindFirstChild("__CHINAESP") then
                        local h = Instance.new("Highlight")
                        h.Name = "__CHINAESP"
                        h.FillColor = Color3.fromRGB(120, 200, 120)
                        h.OutlineColor = Color3.fromRGB(255,255,255)
                        h.FillTransparency = 0.5
                        h.OutlineTransparency = 0
                        h.Parent = part
                        State.Highlights[part] = h
                    end
                end
            end
            task.wait(1)
        end
        -- clean up
        for _, h in pairs(State.Highlights) do
            if h and h.Parent then h:Destroy() end
        end
        State.Highlights = {}
    end)
end

local function setPlayerESP(on)
    task.spawn(function()

            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local h = p.Character:FindFirstChild("__CHINAPLAYESP")

                        h = Instance.new("Highlight")
                        h.Name = "__CHINAPLAYESP"
                        h.FillColor = Color3.fromRGB(220, 60, 60)
                        h.OutlineColor = Color3.fromRGB(255,255,255)
                        h.FillTransparency = 0.6
                        h.OutlineTransparency = 0
                        h.Parent = p.Character
                    end
                end
            end
        else
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local h = p.Character:FindFirstChild("__CHINAPLAYESP")
                    if h then h:Destroy() end
                end
            end
        end
    end)
end

-- ========== SPEED ==========
local speedConn
local function applySpeed(on)
    if speedConn then speedConn:Disconnect() speedConn = nil end
    if not on then return end
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hum = char:WaitForChild("Humanoid")
    local root = char:WaitForChild("HumanoidRootPart")

    -- bypass: keep WalkSpeed = 16, but apply velocity-based boost
    speedConn = RunService.Heartbeat:Connect(function()
        if not Config.SpeedEnabled then return end
        local c = LocalPlayer.Character
        if not c then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        local h = c:FindFirstChild("Humanoid")
        if not r or not h then return end
        if h.MoveDirection.Magnitude > 0 then
            -- apply velocity in move direction (bypasses WalkSpeed check)
            local dir = h.MoveDirection.Unit
            r.AssemblyLinearVelocity = Vector3.new(dir.X * Config.Speed, r.AssemblyLinearVelocity.Y, dir.Z * Config.Speed)
        end
        -- spoof WalkSpeed so server sees 16
        if h.WalkSpeed ~= 16 then h.WalkSpeed = 16 end
    end)
end

-- ========== AUTOFARM LOOPS ==========
local function autoFarmLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoFarm then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local root = char.HumanoidRootPart
                    local eggs = getEggs()
                    for _, egg in ipairs(eggs) do
                        local part = egg:IsA("Model") and (egg.PrimaryPart or egg:FindFirstChildWhichIsA("BasePart")) or egg
                        if part then
                            local d = (root.Position - part.Position).Magnitude
                            if d <= Config.CollectRange then
                                -- touch it
                                firetouchinterest(root, part, 0)
                                task.wait(0.02)
                                firetouchinterest(root, part, 1)
                            elseif d <= 50 then
                                -- walk to it
                                walkTo(part.Position, Config.Speed)
                            end
                        end
                    end
                end
            end
            task.wait(Config.FarmDelay)
        end
    end)
end

local function autoStealLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoSteal then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local root = char.HumanoidRootPart
                    local players = getOtherPlayers()
                    for _, p in ipairs(players) do
                        local target = p.Character:FindFirstChild("HumanoidRootPart")
                        if target then
                            local d = (root.Position - target.Position).Magnitude
                            if d <= Config.StealRange then
                                firetouchinterest(root, target, 0)
                                task.wait(0.02)
                                firetouchinterest(root, target, 1)
                            elseif d <= 60 then
                                walkTo(target.Position, Config.Speed)
                            end
                        end
                    end
                end
            end
            task.wait(Config.StealDelay)
        end
    end)
end

local function autoDepositLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoDeposit then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local root = char.HumanoidRootPart
                    -- only deposit if we're near a deposit zone
                    local zones = getDepositZones()
                    local closest, dist = closestFrom(zones)
                    if closest then
                        local part = closest:IsA("Model") and (closest.PrimaryPart or closest:FindFirstChildWhichIsA("BasePart")) or closest
                        if part and (dist or 999) > 4 then
                            walkTo(part.Position, Config.Speed)
                        elseif part and (dist or 999) <= 4 then
                            firetouchinterest(root, part, 0)
                            task.wait(0.05)
                            firetouchinterest(root, part, 1)
                        end
                    end
                end
            end
            task.wait(Config.DepositDelay)
        end
    end)
end

function serverHop()
    UI.notify("Hopping servers...")
    local existingJson
    pcall(function()
        existingJson = game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
    end)
    if not existingJson then UI.notify("Could not fetch server list") return end

    local decoded = HttpService:JSONDecode(existingJson)
    if not decoded or not decoded.data then UI.notify("Bad server response") return end

    local candidates = {}
    for _, s in ipairs(decoded.data) do
        if s.playing < s.maxPlayers and s.id ~= game.JobId then
            table.insert(candidates, s)
        end
    end

    if #candidates == 0 then UI.notify("No other servers found") return end

    local target = candidates[math.random(1, #candidates)]
    UI.notify("Hopping to server: " .. tostring(target.id))
    TeleportService:TeleportToPlaceInstance(game.PlaceId, target.id, LocalPlayer)
end

-- ========== BOOTSTRAP ==========
local function bootstrap()
    -- wait for character
    if not LocalPlayer.Character then
        LocalPlayer.CharacterAdded:Wait()
    end
    task.wait(0.5)

    -- wire up ESP toggles
    -- (the toggles already call Config update; we need to also trigger the visual system)
    -- we hook into config changes via a small poller:
    task.spawn(function()
        local lastEgg, lastPly
        while State.Running do
            if Config.EggESP ~= lastEgg then
                lastEgg = Config.EggESP
                setEggESP(lastEgg)
            end
            if Config.PlayerESP ~= lastPly then
                lastPly = Config.PlayerESP
                setPlayerESP(lastPly)
            end
            task.wait(0.5)
        end
    end)

    -- wire up speed
    task.spawn(function()
        local last
        while State.Running do
            if Config.SpeedEnabled ~= last then
                last = Config.SpeedEnabled
                applySpeed(last)
            end
            task.wait(0.5)
        end
    end)

    -- start farm loops
    autoFarmLoop()
    autoStealLoop()
    autoDepositLoop()
    autoRejoinLoop()

    State.Loaded = true
end
