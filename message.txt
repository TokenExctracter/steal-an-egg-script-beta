local function tween(obj, props, t, style, dir)
    local info = TweenInfo.new(t or 0.2, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out)
    local tw = TweenService:Create(obj, info, props)
    tw:Play()
    return tw
end

local function lerp(a, b, t) return a + (b - a) * t end

--  REMOTE DETECTION
-- =========================================================
local function scanForRemotes()
    local found = {}
    local function check(container)
        for _, obj in ipairs(container:GetDescendants()) do
            if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                local n = obj.Name:lower()
                if n:match("collect") or n:match("grab") or n:match("pickup") or n:match("getegg") then
                    found.Collect = found.Collect or obj
                elseif n:match("steal") or n:match("take") then
                    found.Steal = found.Steal or obj
                elseif n:match("deposit") or n:match("drop") or n:match("deliver") then
                    found.Deposit = found.Deposit or obj
                elseif n:match("sell") or n:match("cash") or n:match("money") then
                    found.Sell = found.Sell or obj
                elseif n:match("hatch") or n:match("open") then
                    found.Hatch = found.Hatch or obj
                end
            end
        end
    end
    pcall(check, ReplicatedStorage)
    pcall(check, Workspace)
    return found
end

local function fireRemoteSafe(remote, ...)
    if not remote then return false end
    local ok, err = pcall(function(...)
        if remote:IsA("RemoteEvent") then
            remote:FireServer(...)
        elseif remote:IsA("RemoteFunction") then
            remote:InvokeServer(...)
        end
    end, ...)
    return ok
end

local function isValidEgg(obj)
    if not (obj:IsA("BasePart") or obj:IsA("Model")) then return false end
    local n = obj.Name:lower()
    if n:match("esp") or n:match("highlight") or n:match("tracer") or n:match("nametag") then return false end
    if n:match("base") or n:match("zone") or n:match("deposit") or n:match("sell") then return false end
    if obj:IsA("BasePart") and obj.Transparency == 1 and not obj.CanCollide then return false end
    return n:match("egg")
end

local function getEggs()
    local eggs = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if isValidEgg(obj) then
            table.insert(eggs, obj)
        end
    end
    return eggs
end

local function getDepositZones()
    local zones = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local n = obj.Name:lower()
            if n:match("deposit") or n:match("sell") or n:match("zone") or n:match("cashier") or n:match("counter") then
                table.insert(zones, obj)
            end
        end
    end
    return zones
end

local function getOtherPlayers()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            table.insert(list, p)
        end
    end
    return list
end

local function getPart(obj)
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then
        return obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
    end
end

local function closestFrom(list, maxDist)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local root = char.HumanoidRootPart
    local best, bestDist
    for _, obj in ipairs(list) do
        local part = getPart(obj)
        if part then
            local d = (root.Position - part.Position).Magnitude
            if (not maxDist or d <= maxDist) and (not bestDist or d < bestDist) then
                best, bestDist = obj, d
            end
        end
    end
    return best, bestDist
end

--  ANTICHEAT BYPASS

-- WalkSpeed spoofing: hook the WalkSpeed property so any anticheat
-- reading it always sees 16, even when we move faster.
local function installWalkSpeedBypass()
    if State.Hooks.WalkSpeed then return end
    local mt = getrawmetatable(game)
    local oldIndex = mt.__index
    local oldNamecall = mt.__namecall
    setreadonly(mt, false)

    State.Hooks.WalkSpeed = true
    State.Hooks.OldIndex = oldIndex
    State.Hooks.OldNamecall = oldNamecall

    -- Hook __index for WalkSpeed reads
    local newIndex
    newIndex = newcclosure(function(self, key)
        if key == "WalkSpeed" and self:IsA("Humanoid") and self.Parent and self.Parent == LocalPlayer.Character then
            return 16
        end
        return oldIndex(self, key)
    end)
    mt.__index = newIndex

    -- Hook namecall for anticheat-style function calls
    local newNamecall
    newNamecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local args = {...}
        -- Some anticheats call Humanoid:GetPropertyChangedSignal("WalkSpeed") etc — let them through, harmless
        return oldNamecall(self, ...)
    end)
    mt.__namecall = newNamecall

    setreadonly(mt, true)
end

-- Velocity spoofing: apply movement velocity only on Heartbeat,
-- restore on Stepped so server-queried values during physics step
-- don't show boosted values.
local function applySpeedBypass()
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not Config.SpeedEnabled then return end
        local c = LocalPlayer.Character
        if not c then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        local h = c:FindFirstChild("Humanoid")
        if not r or not h then return end
        if h.MoveDirection.Magnitude > 0 then
            local dir = h.MoveDirection.Unit
            r.AssemblyLinearVelocity = Vector3.new(dir.X * Config.Speed, r.AssemblyLinearVelocity.Y, dir.Z * Config.Speed)
        end
    end)
    return conn
end

-- Anti-knockback: zero out velocity changes from server
local function applyAntiKnockback()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if not Config.AntiKnockback then return end
        local c = LocalPlayer.Character
        if not c then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        if not r then return end
        local v = r.AssemblyLinearVelocity
        -- If we have unexpected horizontal velocity while not moving, zero it
        if v.X ~= 0 or v.Z ~= 0 then
            local h = c:FindFirstChild("Humanoid")
            if h and h.MoveDirection.Magnitude == 0 then
                r.AssemblyLinearVelocity = Vector3.new(0, v.Y, 0)
            end
        end
    end)
    return conn
end

-- =========================================================
--  MOVEMENT (PATHFINDING-BASED, NO TELEPORT FLAGS)
-- =========================================================
local function walkTo(targetPos, speed)
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum  = char:FindFirstChild("Humanoid")
    if not root or not hum then return end

    if Config.UsePathfind then
        -- PathfindingService gives legit-looking movement
        local path = PathfindingService:CreatePath({
            AgentRadius = 2,
            AgentHeight = 5,
            AgentCanJump = true,
            AgentCanClimb = false,
            WaypointSpacing = 4,
            Costs = { Water = 20 },
        })
        local ok = pcall(function()
            path:ComputeAsync(root.Position, targetPos)
        end)
        if ok and path.Status == Enum.PathStatus.Success then
            for _, wp in ipairs(path:GetWaypoints()) do
                if not State.Running then return end
                hum:MoveTo(wp.Position)
                if wp.Action == Enum.PathWaypointAction.Jump then
                    hum.Jump = true
                end
                -- wait until reached or timeout
                local start = tick()
                while (root.Position - wp.Position).Magnitude > 4 and tick() - start < 3 do
                    task.wait(0.05)
                end
            end
        else
            -- fallback to direct walk
            hum:MoveTo(targetPos)
            local start = tick()
            while (root.Position - Vector3.new(targetPos.X, root.Position.Y, targetPos.Z)).Magnitude > 3 and tick() - start < 5 do
                task.wait(0.05)
            end
        end
    else
        hum:MoveTo(targetPos)
        local start = tick()
        while (root.Position - Vector3.new(targetPos.X, root.Position.Y, targetPos.Z)).Magnitude > 3 and tick() - start < 5 do
            task.wait(0.05)
        end
    end
end

local function teleportTo(pos)
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    root.CFrame = root.CFrame + (pos - root.Position)
end

local flyConn, flyBV, flyBG
local function startFly()
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:WaitForChild("HumanoidRootPart", 3)
    if not root then return end
    if flyBV then flyBV:Destroy() end
    if flyBG then flyBG:Destroy() end
    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBV.Velocity = Vector3.new(0,0,0)
    flyBV.Parent = root
    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBG.P = 10000
    flyBG.CFrame = root.CFrame
    flyBG.Parent = root

    local cam = Workspace.CurrentCamera
    flyConn = RunService.RenderStepped:Connect(function()
        if not Config.FlyEnabled then return end
        local c = LocalPlayer.Character
        if not c then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        if not r then return end
        local fwd, side, up = 0, 0, 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then fwd = 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then fwd = -1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then side = -1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then side = 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then up = 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then up = -1 end
        local moveVec = cam.CFrame:VectorToWorldSpace(Vector3.new(side, 0, -fwd))
        local vel = moveVec * Config.FlySpeed + Vector3.new(0, up * Config.FlySpeed, 0)
        flyBV.Velocity = vel
        flyBG.CFrame = cam.CFrame
    end)
end

local function stopFly()
    if flyConn then flyConn:Disconnect() flyConn = nil end
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
end

-- =========================================================
--  COLLECTION LOGIC (TOUCH + REMOTE FALLBACK)
-- =========================================================
local function tryCollectEgg(egg)
    local part = getPart(egg)
    if not part then return false end
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return false end

    -- Method 1: Remote (most reliable)
    if State.Remote.Collect then
        local ok = fireRemoteSafe(State.Remote.Collect, part)
        if ok then return true end
    end

    -- Method 2: Touch interest
    pcall(function()
        firetouchinterest(root, part, 0)
        task.wait(0.02)
        firetouchinterest(root, part, 1)
    end)

    -- Method 3: Some games use a ProximityPrompt
    local prompt = part:FindFirstChildWhichIsA("ProximityPrompt") or (egg:IsA("Model") and egg:FindFirstChildWhichIsA("ProximityPrompt"))
    if prompt then
        pcall(function() fireproximityprompt(prompt, 0) end)
    end

    return true
end

local function tryStealFromPlayer(p)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local target = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
    if not target then return end

    -- Method 1: Remote
    if State.Remote.Steal then
        fireRemoteSafe(State.Remote.Steal, p)
    end

    -- Method 2: Touch
    pcall(function()
        firetouchinterest(root, target, 0)
        task.wait(0.02)
        firetouchinterest(root, target, 1)
    end)
end

local function tryDeposit(zone)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local part = getPart(zone)
    if not part then return end

    -- Method 1: Remote
    if State.Remote.Deposit then
        fireRemoteSafe(State.Remote.Deposit)
    elseif State.Remote.Sell then
        fireRemoteSafe(State.Remote.Sell)
    end

    -- Method 2: Touch
    pcall(function()
        firetouchinterest(root, part, 0)
        task.wait(0.05)
        firetouchinterest(root, part, 1)
    end)

    -- Method 3: ProximityPrompt
    local prompt = part:FindFirstChildWhichIsA("ProximityPrompt")
    if prompt then
        pcall(function() fireproximityprompt(prompt, 0) end)
    end
end

local function tryHatch()
    if State.Remote.Hatch then
        fireRemoteSafe(State.Remote.Hatch)
    end
end

-- =========================================================
--  AUTO LOOPS
-- =========================================================
local function autoFarmLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoFarm then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local root = char.HumanoidRootPart
                    local eggs = getEggs()
                    local closest, dist = closestFrom(eggs, 200)
                    if closest then
                        local part = getPart(closest)
                        if part then
                            if dist <= Config.CollectRange then
                                tryCollectEgg(closest)
                            else
                                walkTo(part.Position, Config.Speed)
                            end
                        end
                    end
                end
            end
            task.wait(Config.FarmDelay)
        end
    end)
end

local function autoStealLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoSteal then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local players = getOtherPlayers()
                    local closest, dist = closestFrom(players, 200)
                    if closest and closest.Character then
                        local target = closest.Character:FindFirstChild("HumanoidRootPart")
                        if target then
                            if dist <= Config.StealRange then
                                tryStealFromPlayer(closest)
                            else
                                walkTo(target.Position, Config.Speed)
                            end
                        end
                    end
                end
            end
            task.wait(Config.StealDelay)
        end
    end)
end

local function autoDepositLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoDeposit then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local zones = getDepositZones()
                    local closest, dist = closestFrom(zones, 200)
                    if closest then
                        local part = getPart(closest)
                        if part then
                            if dist and dist <= 5 then
                                tryDeposit(closest)
                            else
                                walkTo(part.Position, Config.Speed)
                            end
                        end
                    end
                end
            end
            task.wait(Config.DepositDelay)
        end
    end)
end

local function autoHatchLoop()
    task.spawn(function()
        while State.Running do
            if Config.AutoHatch then tryHatch() end
            task.wait(0.5)
        end
    end)
end

local function autoRejoinLoop()
    task.spawn(function()
        while State.Running do
            task.wait(5)
            if Config.AutoRejoin then
                local conns = NetworkClient:GetConnections()
                if not conns or #conns == 0 then
                    pcall(function() TeleportService:Teleport(game.PlaceId, LocalPlayer) end)
                end
            end
        end
    end)
end

local tracerConn
local function setTracers(on)
    if tracerConn then tracerConn:Disconnect() tracerConn = nil end
    if not on then
        for _, a in pairs(State.TracersAdorn) do if a and a.Parent then a:Destroy() end end
        State.TracersAdorn = {}
        return
    end
    tracerConn = RunService.RenderStepped:Connect(function()
        local char = LocalPlayer.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        local cam = Workspace.CurrentCamera
        local root = char.HumanoidRootPart
        -- clear old
        for _, a in pairs(State.TracersAdorn) do if a and a.Parent then a:Destroy() end end
        State.TracersAdorn = {}
        if Config.EggESP then
            for _, egg in ipairs(getEggs()) do
                local part = getPart(egg)
                if part then
                    local dist = (root.Position - part.Position).Magnitude
                    if dist < 200 then
                        local adorn = Instance.new("Beam")
                        adorn.FaceCamera = true
                        adorn.Width0 = 0.1
                        adorn.Width1 = 0.1
                        adorn.Color = ColorSequence.new(Color3.fromRGB(120, 220, 120))
                        adorn.Transparency = NumberSequence.new(0.3, 0.8)
                        local a0 = Instance.new("Attachment", root)
                        local a1 = Instance.new("Attachment", part)
                        adorn.Attachment0 = a0
                        adorn.Attachment1 = a1
                        adorn.Parent = Workspace.CurrentCamera
                        table.insert(State.TracersAdorn, adorn)
                        table.insert(State.TracersAdorn, a0)
                        table.insert(State.TracersAdorn, a1)
                    end
                end
            end
        end
    end)
end

                    s.Parent = store
                    store.Parent = part
                    part.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize)
                end
            end
        end
    end)
end

-- =========================================================
--  CLICK TP
-- =========================================================
local clickConn
local function setClickTP(on)
    if clickConn then clickConn:Disconnect() clickConn = nil end
    if not on then return end
    clickConn = UserInputService.InputBegan:Connect(function(i, gp)
        if gp then return end
        if i.UserInputType == Enum.UserInputType.MouseButton1 and UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
            local mouse = LocalPlayer:GetMouse()
            if mouse.Hit then
                teleportTo(mouse.Hit.Position)
            end
        end
    end)
end

-- =========================================================
--  CONFIG SAVE / LOAD
-- =========================================================
local configFolder = "ChinaV4Egg"
function saveConfig()
    if not writefile then UI.notify("writefile not supported") return end
    if not isfolder(configFolder) then makefolder(configFolder) end
    local data = HttpService:JSONEncode(Config)
    writefile(configFolder .. "/config.json", data)
    UI.notify("Config saved")
end

function loadConfig()
    if not readfile then UI.notify("readfile not supported") return end
    if not isfile(configFolder .. "/config.json") then UI.notify("No saved config") return end
    local data = readfile(configFolder .. "/config.json")
    local loaded = HttpService:JSONDecode(data)
    for k, v in pairs(loaded) do Config[k] = v end
    UI.notify("Config loaded")
end

-- =========================================================
--  SERVER HOP / REJOIN
-- =========================================================
function rejoin()
    UI.notify("Rejoining...")
    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer) end)
end

function serverHop()
    UI.notify("Hopping servers...")
    local json
    pcall(function() json = game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100") end)
    if not json then UI.notify("Could not fetch servers") return end
    local decoded = HttpService:JSONDecode(json)
    if not decoded or not decoded.data then UI.notify("Bad response") return end
    local cands = {}
    for _, s in ipairs(decoded.data) do
        if s.playing < s.maxPlayers and s.id ~= game.JobId then table.insert(cands, s) end
    end
    if #cands == 0 then UI.notify("No other servers") return end
    local target = cands[math.random(1, #cands)]
    UI.notify("Hopping to: " .. tostring(target.id):sub(1, 8) .. "...")
    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, target.id, LocalPlayer) end)
end
